const handleDownloadFile = async (node) => {
if (!node?.storagePath) return;

// Trigger the actual browser download (works even for cross-origin URLs,
// unlike the plain <a download> attribute).
try {
  const response = await fetch(node.storagePath);
  if (!response.ok) throw new Error("Failed to fetch file");
  const blob = await response.blob();
  const blobUrl = window.URL.createObjectURL(blob);

  const a = document.createElement("a");
  a.href = blobUrl;
  a.download = node.originalFileName || node.name || "download";
  document.body.appendChild(a);
  a.click();
  a.remove();
  window.URL.revokeObjectURL(blobUrl);
} catch (err) {
  setErrorMsg("Failed to download file.");
  return; // don't log a download that didn't actually happen
}

// Log the download  fire-and-forget, don't let a logging hiccup bother the user
try {
  await callApi("DOWNLOADFILE", { NodeId: node.id });
} catch {
  // silent
}
};
