            {snackbar.message}
          </Alert>
        </Snackbar>
      </Box>
    </LocalizationProvider>
  );
};

export default OrderForm;

    try {
      setIsSubmitting(true);
      const OrderMail = generateOrderEmail(formData, diamondOptions);
      const CustomerConfirmationEmail = generateCustomerConfirmationEmail(formData, diamondOptions);
      
      await sendEmail({
        subject: `New Customize Order Request received - ${formData?.name}`,
        cust_subject: `Customize Order Request Has Been placed - Elior Jewels`,
        attachments: file ? [file] : [],
        replyto: formData.email,
        Mails: storeInit?.Website_Email,
        CustomerMail: formData?.email,
        htmlTemplate: OrderMail,
        cust_htmlTemplate: CustomerConfirmationEmail,
      });

      setSnackbar({ open: true, message: "Order submitted successfully!", severity: "success" });
      resetForm();
    } catch (error) {
      console.error("Submit error:", error);
      setSnackbar({ open: true, message: "Something went wrong. Please try again.", severity: "error" });
    } finally {
      setIsSubmitting(false);
    }
  };

  useEffect(() => {
    if (snackbar.open && snackbar.severity === "success") {
      window.scrollTo({ top: 0, behavior: "smooth" });
    }
  }, [snackbar]);

  // UI Component for Section Headers
  const SectionTitle = ({ title, step }) => (
    <Box sx={{ display: 'flex', alignItems: 'center', mb: 3, mt: 2 }}>
      <Box sx={{ 
        bgcolor: COLORS.gold, color: '#fff', borderRadius: '50%', width: 28, height: 28, 
        display: 'flex', justifyContent: 'center', alignItems: 'center', mr: 1.5, fontSize: '0.85rem', fontWeight: 'bold' 
      }}>
        {step}
      </Box>
      <Typography variant="h6" sx={{ fontWeight: 700, color: COLORS.black }}>{title}</Typography>
    </Box>
  );

  return (
    <ThemeProvider theme={theme}>
      <LocalizationProvider dateAdapter={AdapterDayjs}>
        <Box sx={{ backgroundColor: COLORS.bg, minHeight: "100vh", pt: { xs: 2, sm: 4 }, pb: { xs: 10, sm: 12 } }}>
          <Container maxWidth="md">
            
            {/* --- BRANDING BANNER --- */}
            <Box sx={{ 
              backgroundColor: COLORS.black, borderRadius: { xs: 3, sm: 4 }, p: { xs: 3, sm: 4 }, 
              mb: 3, textAlign: "center", boxShadow: '0 8px 24px rgba(0,0,0,0.12)' 
            }}>
              {companyLogo ? (
                <Box component="img" src={companyLogo} alt="Elior Jewel" sx={{ maxHeight: 60, objectFit: "contain", mb: 2 }} />
              ) : (
                <Typography variant="h4" sx={{ color: COLORS.gold, fontWeight: 'bold', mb: 2 }}>Elior Jewel</Typography>
              )}
              <Typography variant="h6" sx={{ color: '#fff', fontWeight: 600, textTransform: "uppercase", letterSpacing: 1 }}>
                Custom Order Request
              </Typography>
              <Typography variant="body2" sx={{ color: '#aaa', mt: 1, maxWidth: 500, mx: 'auto' }}>
                Fill out the details below to place your custom order. Ensure all specifications are accurate.
              </Typography>
              <Stack direction="row" flexWrap="wrap" justifyContent="center" gap={1} sx={{ mt: 3 }}>
                {CONTACT_NUMBERS.map((num, i) => (
                  <Chip key={i} label={num} size="small" sx={{ bgcolor: 'rgba(255,255,255,0.1)', color: COLORS.gold, border: `1px solid ${COLORS.gold}` }} />
                ))}
              </Stack>
            </Box>

            <Paper elevation={0} sx={{ borderRadius: { xs: 3, sm: 4 }, overflow: "hidden", boxShadow: '0 4px 24px rgba(0,0,0,0.04)', border: `1px solid ${COLORS.border}` }}>
              
              <Box sx={{ p: { xs: 2.5, sm: 5 } }}>
                <Stack spacing={4}>
                  
                  {/* --- SECTION 1: CONTACT INFO --- */}
                  <Box>
                    <SectionTitle step="1" title="Contact Information" />
                    <Stack spacing={2.5}>
                      <TextField label="Full Name" name="name" value={formData.name} onChange={handleInputChange} error={!!errors.name} helperText={errors.name} required placeholder="Enter your full name" />
                      <Stack direction={{ xs: "column", sm: "row" }} spacing={2.5}>
                        <TextField label="Email Address" type="email" name="email" value={formData.email} onChange={handleInputChange} error={!!errors.email} helperText={errors.email} required placeholder="Enter your Email" />
                        <TextField label="Mobile Number" type="tel" name="mobile" value={formData.mobile} onChange={handleInputChange} error={!!errors.mobile} helperText={errors.mobile} required placeholder="Enter your Mobile" />
                      </Stack>
                    </Stack>
                  </Box>

                  <Divider sx={{ borderColor: COLORS.border }} />

                  {/* --- SECTION 2: PRODUCT DETAILS --- */}
                  <Box>
                    <SectionTitle step="2" title="Product Details" />
                    <Stack spacing={3}>
                      
                      {/* Image Upload Mobile Friendly Dropzone */}
                      <FormControl fullWidth required error={!!errors.file}>
                        <FormLabel sx={{ mb: 1, fontWeight: 600, color: COLORS.black }}>Reference Image</FormLabel>
                        {!file ? (
                          <Button 
                            component="label" 
                            sx={{ 
                              border: `2px dashed ${errors.file ? COLORS.error : '#c4cdd5'}`, 
                              bgcolor: '#f9fafb', height: 120, display: 'flex', flexDirection: 'column', 
                              color: COLORS.textGray, borderRadius: 3, '&:hover': { bgcolor: '#f4f6f8', borderColor: COLORS.gold }
                            }}
                          >
                            <CloudUploadOutlinedIcon sx={{ fontSize: 40, mb: 1, color: COLORS.gold }} />
                            <Typography variant="body2" fontWeight={600}>Tap to upload product image</Typography>
                            <Typography variant="caption" color="textSecondary">Max size: 5MB (JPEG, PNG)</Typography>
                            <input ref={fileInputRef} hidden accept="image/*" type="file" onChange={handleFileChange} />
                          </Button>
                        ) : (
                          <Box sx={{ display: "flex", alignItems: "center", justifyContent: 'space-between', p: 2, border: `1px solid ${COLORS.gold}`, borderRadius: 3, bgcolor: '#fffdf9' }}>
                            <Box sx={{ display: 'flex', alignItems: 'center', gap: 1.5, overflow: 'hidden' }}>
                              <CheckCircleOutlineIcon sx={{ color: COLORS.gold }} />
                              <Typography variant="body2" sx={{ fontWeight: 600, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>
                                {file.name}
                              </Typography>
                            </Box>
                            <Button size="small" color="error" onClick={clearFile} sx={{ minWidth: 'auto', p: 1 }}>
                              <CloseIcon fontSize="small" />
                            </Button>
                          </Box>
                        )}
                        {errors.file && <FormHelperText error>{errors.file}</FormHelperText>}
                      </FormControl>

                      <TextField label="Design Number" name="designNumber" value={formData.designNumber} onChange={handleInputChange} error={!!errors.designNumber} helperText={errors.designNumber} required />
                      
                      <Box>
                        <TextField multiline minRows={2} label="Product Size" name="productSize" value={formData.productSize} onChange={handleInputChange} error={!!errors.productSize} helperText={errors.productSize} required placeholder="E.g., Ring Size 7, 18-inch Chain..." />
                        <FormHelperText sx={{ ml: 0.5 }}>Please mention Ring, Bracelet, or Chain sizes here.</FormHelperText>
                      </Box>

                      <Stack direction={{ xs: "column", sm: "row" }} spacing={2.5}>
                        <TextField select label="Color of Product" name="color" value={formData.color} onChange={handleInputChange} error={!!errors.color} helperText={errors.color} required>
                          {OPTIONS.colors.map((opt) => <MenuItem key={opt} value={opt}>{opt}</MenuItem>)}
                        </TextField>
                        <TextField select label="Karats of Product" name="karats" value={formData.karats} onChange={handleInputChange} error={!!errors.karats} helperText={errors.karats} required>
                          {OPTIONS.karats.map((opt) => <MenuItem key={opt} value={opt}>{opt}</MenuItem>)}
                        </TextField>
                      </Stack>
                    </Stack>
                  </Box>

                  <Divider sx={{ borderColor: COLORS.border }} />

                  {/* --- SECTION 3: CUSTOMIZATIONS --- */}
                  <Box>
                    <SectionTitle step="3" title="Customizations" />
                    <Stack spacing={3}>
                      
                      {/* Rhodium & Stamping */}
                      <Grid container spacing={3}>
                        <Grid item xs={12} sm={6}>
                          <FormControl required error={!!errors.rhodium} sx={{ width: '100%' }}>
                            <FormLabel sx={{ fontWeight: 600, color: COLORS.black, mb: 1 }}>Rhodium Plating</FormLabel>
                            <RadioGroup name="rhodium" value={formData.rhodium} onChange={handleInputChange}>
                              {OPTIONS.rhodium.map((opt) => (
                                <FormControlLabel key={opt} value={opt} control={<Radio />} label={<Typography variant="body2">{opt}</Typography>} sx={{ mb: 0.5 }} />
                              ))}
                            </RadioGroup>
                            {formData.rhodium === "Other" && (
                              <TextField size="small" placeholder="Specify rhodium details" name="otherRhodium" value={formData.otherRhodium} onChange={handleInputChange} error={!!errors.rhodium && !formData.otherRhodium} sx={{ mt: 1 }} />
                            )}
                          </FormControl>
                        </Grid>

                        <Grid item xs={12} sm={6}>
                          <FormControl required error={!!errors.stamping} sx={{ width: '100%' }}>
                            <FormLabel sx={{ fontWeight: 600, color: COLORS.black, mb: 1 }}>Stamping</FormLabel>
                            <RadioGroup name="stamping" value={formData.stamping} onChange={handleInputChange}>
                              {OPTIONS.stamping.map((opt) => (
                                <FormControlLabel key={opt} value={opt} control={<Radio />} label={<Typography variant="body2">{opt}</Typography>} sx={{ mb: 0.5 }} />
                              ))}
                            </RadioGroup>
                            {formData.stamping === "Other" && (
                              <TextField size="small" placeholder="Specify stamping details" name="otherStamping" value={formData.otherStamping} onChange={handleInputChange} error={!!errors.stamping && !formData.otherStamping} sx={{ mt: 1 }} />
                            )}
                          </FormControl>
                        </Grid>
                      </Grid>

                      {/* Stones & Diamonds */}
                      <Box sx={{ bgcolor: '#f9fafb', p: 2, borderRadius: 3, border: `1px solid ${COLORS.border}` }}>
                        <FormControl component="fieldset" fullWidth>
                          <FormLabel sx={{ fontWeight: 600, color: COLORS.black, mb: 1 }}>Diamonds & Stones</FormLabel>
                          <Typography variant="caption" color="textSecondary" sx={{ mb: 1, display: "block" }}>
                            Tick only if items need to be sourced by Elior Jewel.
                          </Typography>
                          <FormGroup>
                            <Grid container spacing={1}>
                              <Grid item xs={6} sm={3}><FormControlLabel control={<Checkbox name="diamond" checked={diamondOptions.diamond} onChange={handleCheckboxChange} />} label={<Typography variant="body2">Diamond</Typography>} /></Grid>
                              <Grid item xs={6} sm={3}><FormControlLabel control={<Checkbox name="colorStone" checked={diamondOptions.colorStone} onChange={handleCheckboxChange} />} label={<Typography variant="body2">Color Stone</Typography>} /></Grid>
                              <Grid item xs={6} sm={3}><FormControlLabel control={<Checkbox name="byParty" checked={diamondOptions.byParty} onChange={handleCheckboxChange} />} label={<Typography variant="body2">By Party</Typography>} /></Grid>
                              <Grid item xs={6} sm={3}><FormControlLabel control={<Checkbox name="other" checked={diamondOptions.other} onChange={handleCheckboxChange} />} label={<Typography variant="body2">Other</Typography>} /></Grid>
                            </Grid>
                          </FormGroup>
                        </FormControl>
                      </Box>
                    </Stack>
                  </Box>

                  <Divider sx={{ borderColor: COLORS.border }} />

                  {/* --- SECTION 4: DELIVERY & SUBMIT --- */}
                  <Box>
                    <SectionTitle step="4" title="Delivery & Instructions" />
                    <Stack spacing={3}>
                      
                      <Grid container spacing={2.5}>
                        <Grid item xs={12} sm={6}>
                          <DatePicker label="Preferred Delivery Date" value={formData.deliveryDate} onChange={(val) => handleDateChange("deliveryDate", val)} slotProps={{ textField: { fullWidth: true } }} />
                        </Grid>
                        <Grid item xs={12} sm={6}>
                          <TimePicker label="Preferred Delivery Time" value={formData.deliveryTime} onChange={(val) => handleDateChange("deliveryTime", val)} slotProps={{ textField: { fullWidth: true } }} />
                        </Grid>
                      </Grid>

                      <TextField label="Additional Instructions" multiline minRows={3} fullWidth name="instructions" value={formData.instructions} onChange={handleInputChange} placeholder="Any specific requests or instructions for production..." />

                      <Box sx={{ mt: 2 }}>
                        <Button
                          variant="contained"
                          color="success"
                          size="large"
                          fullWidth
                          onClick={handleSubmit}
                          disabled={isSubmitting}
                          sx={{
                            py: 1.8,
                            fontSize: "1.1rem",
                            borderRadius: 3,
                            boxShadow: '0 8px 16px rgba(30, 188, 87, 0.24)',
                            transition: 'all 0.3s',
                            "&:hover": { 
                              backgroundColor: "#179e49",
                              boxShadow: '0 8px 20px rgba(30, 188, 87, 0.4)'
                            },
                          }}
                        >
                          {isSubmitting ? "Processing Order..." : "Submit Order Securely"}
                        </Button>
                      </Box>
                    </Stack>
                  </Box>

                </Stack>
              </Box>
            </Paper>
          </Container>

          {/* --- FEEDBACK --- */}
          <Snackbar open={snackbar.open} autoHideDuration={5000} onClose={() => setSnackbar({ ...snackbar, open: false })} anchorOrigin={{ vertical: "bottom", horizontal: "center" }}>
            <Alert variant="filled" onClose={() => setSnackbar({ ...snackbar, open: false })} severity={snackbar.severity} sx={{ width: "100%", borderRadius: 2 }}>
              {snackbar.message}
            </Alert>
          </Snackbar>
        </Box>
      </LocalizationProvider>
    </ThemeProvider>
  );
};

export default OrderForm;
